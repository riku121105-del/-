<script>
    function showToast(message) {
        const toast = document.getElementById('toast');
        toast.textContent = message;
        toast.classList.add('show');
        setTimeout(() => {
            toast.classList.remove('show');
        }, 2500);
    }

    function selectStyle(element) {
        document.querySelectorAll('.style-card').forEach(c => c.classList.remove('active'));
        element.classList.add('active');
        const name = element.querySelector('.name').textContent;
        document.getElementById('styleLabel').textContent = 'スタイル: ' + name + '調';
    }

    function startGeneration() {
        showToast('✨ アニメーション生成を開始しました！');
        let progress = 0;
        const interval = setInterval(() => {
            progress += 10;
            if (progress >= 100) {
                clearInterval(interval);
                showToast('🎉 アニメーション生成が完了しました！');
            }
        }, 300);
    }

    // Canvas アニメーション描画
    const canvas = document.getElementById('animeCanvas');
    const ctx = canvas.getContext('2d');
    let frame = 0;

    function resizeCanvas() {
        canvas.width = canvas.clientWidth;
        canvas.height = 220;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    function draw() {
        frame++;
        ctx.fillStyle = '#0f0c24';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        // 背景の星・光の粒子
        for (let i = 0; i < 25; i++) {
            let x = (Math.sin(i * 99 + frame * 0.01) * 0.5 + 0.5) * canvas.width;
            let y = (Math.cos(i * 33 + frame * 0.015) * 0.5 + 0.5) * canvas.height;
            let radius = (i % 3) + 1;
            ctx.fillStyle = i % 2 === 0 ? 'rgba(255, 51, 102, 0.7)' : 'rgba(255, 153, 51, 0.7)';
            ctx.beginPath();
            ctx.arc(x, y, radius, 0, Math.PI * 2);
            ctx.fill();
        }

        // キャラクター（簡易セル画風シルエット）の描画
        let cx = canvas.width / 2;
        let cy = 130 + Math.sin(frame * 0.05) * 3;

        // 髪の毛
        ctx.fillStyle = '#2b1b40';
        ctx.beginPath();
        ctx.arc(cx, cy - 20, 35, 0, Math.PI * 2);
        ctx.fill();

        // 顔
        ctx.fillStyle = '#ffdfd3';
        ctx.beginPath();
        ctx.arc(cx, cy - 15, 26, 0, Math.PI * 2);
        ctx.fill();

        // 目
        ctx.fillStyle = '#ff3366';
        ctx.beginPath();
        ctx.arc(cx - 9, cy - 18, 4, 0, Math.PI * 2);
        ctx.arc(cx + 9, cy - 18, 4, 0, Math.PI * 2);
        ctx.fill();

        // 服
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.moveTo(cx - 30, cy + 50);
        ctx.lineTo(cx + 30, cy + 50);
        ctx.lineTo(cx + 20, cy + 10);
        ctx.lineTo(cx - 20, cy + 10);
        ctx.closePath();
        ctx.fill();

        requestAnimationFrame(draw);
    }
    draw();
</script>
